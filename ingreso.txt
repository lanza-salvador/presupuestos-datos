'========================================================================
' MÓDULO DE CÓDIGO: ModIngreso
' Opciones de ingreso a PSA: lo que invierte una persona para empezar (Kit de Acceso + equipos),
' al contado y en 6 y 12 cuotas, con monotributo y sin monotributo.
'   - QUÉ LLEVA cada opción viene de ingreso.txt (GitHub): se cambia sin tocar la app
'   - Precios: costo de DJ de la lista de PSA (ModCostos), porque quien entra es DJ
'   - Percepción de Ingresos Brutos: la carga el distribuidor en la pantalla (la que le cobra PSA),
'     % sobre el precio de lista sin IVA. Depende de la provincia y de la inscripción de cada uno,
'     y cambia: por eso no va fija en ningún lado. En 0, la hoja aclara que no la incluye.
'   - PB: suma de los PB de los equipos (el kit no tiene).
'     9 PB o más: ingreso desarrollado · 13 PB o más: entra como D
'   - Valor de venta: precios sugeridos de los equipos (sin el kit)
' La hoja (A4 parada, 794 x 1123 px) se arma en HTML y se muestra con PageDocumento.
'========================================================================

Sub Process_Globals
	Public Const ARCHIVO As String = "ingreso.txt"
	Type OpcionIngreso (Titulo As String, Destacada As Boolean, Items As List)
	Type ItemIngreso (Cantidad As Int, Producto As String, Nombre As String)
	Type CuentaIngreso (Contado As Double, ContadoNoCat As Double, PuntosPB As Double, Venta As Double, Falta As String)
	Private mapaKits As Map            ' "Portátil" → "KIT DE ACCESO C/PORTATIL ARG."
	Private listaOpciones As List      ' de OpcionIngreso, en el orden del archivo
	Private notaArchivo As String      ' texto de abajo de la hoja (si ingreso.txt trae NOTA)
	Private Const KIT_INICIAL As String = "Portátil"
End Sub

' Al abrir la app: lo que quedó guardado de la última vez
Public Sub Inicializar
	mapaKits.Initialize
	listaOpciones.Initialize
	notaArchivo = ""
	If File.Exists(File.DirInternal, ARCHIVO) Then Interpretar(File.ReadString(File.DirInternal, ARCHIVO))
End Sub

' Lo llama B4XMainPage con ingreso.txt recién bajado. Devuelve un texto para el Log.
Public Sub Importar (Texto As String) As String
	If Interpretar(Texto) = False Then Return "ingreso.txt vacío o con otro formato (se usa el guardado)"
	File.WriteString(File.DirInternal, ARCHIVO, Texto)
	Return mapaKits.Size & " kits · " & listaOpciones.Size & " opciones"
End Sub

' Solo reemplaza lo que había si el archivo trae al menos un kit y una opción
Private Sub Interpretar (Texto As String) As Boolean
	Dim k As Map
	k.Initialize
	Dim ops As List
	ops.Initialize
	Dim n As String = ""
	For Each linea As String In Regex.Split("\r?\n", Texto)
		Dim t As String = linea.Trim
		If t = "" Or t.StartsWith("#") Then Continue
		Dim p() As String = Regex.Split(";", t)
		If p.Length < 2 Then Continue
		Dim tipo As String = p(0).Trim.ToUpperCase
		If tipo = "KIT" And p.Length >= 3 Then
			k.Put(p(1).Trim, p(2).Trim)
		Else If tipo = "OPCION" And p.Length >= 3 Then
			Dim o As OpcionIngreso = LeerOpcion(p(1), p(2))
			If o.Items.Size > 0 Then ops.Add(o)
		Else If tipo = "NOTA" Then
			n = t.SubString(t.IndexOf(";") + 1).Trim
		End If
	Next
	If k.Size = 0 Or ops.Size = 0 Then Return False
	mapaKits = k
	listaOpciones = ops
	notaArchivo = n
	Return True
End Sub

' "*Promocionado" y "1 x PSA VERO BIANCO + KIT POSV. = Vero Bianco|1 x ..." → OpcionIngreso
Private Sub LeerOpcion (Titulo As String, Productos As String) As OpcionIngreso
	Dim o As OpcionIngreso
	o.Initialize
	o.Items.Initialize
	o.Titulo = Titulo.Trim
	o.Destacada = False
	If o.Titulo.StartsWith("*") Then
		o.Destacada = True
		o.Titulo = o.Titulo.SubString(1).Trim
	End If
	For Each parte As String In Regex.Split("\|", Productos)
		Dim t As String = parte.Trim
		If t = "" Then Continue
		Dim it As ItemIngreso
		it.Initialize
		it.Cantidad = 1
		it.Nombre = ""
		Dim igual As Int = t.IndexOf(" = ")
		If igual > 0 Then
			it.Nombre = t.SubString(igual + 3).Trim
			t = t.SubString2(0, igual).Trim
		End If
		Dim m As Matcher = Regex.Matcher2("^(\d+)\s*x\s+(.+)$", Regex.CASE_INSENSITIVE, t)
		If m.Find Then
			it.Cantidad = m.Group(1)
			t = m.Group(2).Trim
		End If
		it.Producto = t
		If it.Nombre = "" Then it.Nombre = ModEquipos.NombreProlijo(t)
		If it.Cantidad > 0 Then o.Items.Add(it)
	Next
	Return o
End Sub

'========================================================================
' PARA LA PANTALLA (PageIngreso)
'========================================================================

Public Sub HayOpciones As Boolean
	Return mapaKits.Size > 0 And listaOpciones.Size > 0
End Sub

' Los kits que están hoy en la lista de PSA ("Portátil", "Mini")
Public Sub Kits As List
	Dim l As List
	l.Initialize
	For Each k As String In mapaKits.Keys
		If ModCostos.Costo(mapaKits.Get(k), "DJ", False) > 0 Then l.Add(k)
	Next
	Return l
End Sub

Public Sub KitInicial As String
	Dim l As List = Kits
	If l.IndexOf(KIT_INICIAL) >= 0 Then Return KIT_INICIAL
	If l.Size > 0 Then Return l.Get(0)
	Return ""
End Sub

' "5 opciones. Al contado, con monotributo: de $ 930.490 a $ 3.297.250"
Public Sub Resumen (Kit As String, Percepcion As Double) As String
	Dim minimo As Double = 0
	Dim maximo As Double = 0
	Dim n As Int = 0
	For Each o As OpcionIngreso In listaOpciones
		Dim c As CuentaIngreso = Calcular(o, Kit, Percepcion)
		If c.Falta <> "" Then Continue
		If n = 0 Or c.Contado < minimo Then minimo = c.Contado
		If c.Contado > maximo Then maximo = c.Contado
		n = n + 1
	Next
	If n = 0 Then Return ""
	Return n & " opciones. Al contado, con monotributo:" & CRLF & "de " & ModAhorro.PesosTexto(minimo) & " a " & ModAhorro.PesosTexto(maximo)
End Sub

' Productos de ingreso.txt que no están en la lista de PSA (esas opciones no van en la hoja)
Public Sub Faltantes As String
	Dim vistos As Map
	vistos.Initialize
	For Each o As OpcionIngreso In listaOpciones
		For Each it As ItemIngreso In o.Items
			If ModCostos.Costo(it.Producto, "DJ", False) <= 0 Then vistos.Put(it.Producto, "")
		Next
	Next
	Dim sb As StringBuilder
	sb.Initialize
	For Each k As String In vistos.Keys
		If sb.Length > 0 Then sb.Append(", ")
		sb.Append(k)
	Next
	Return sb.ToString
End Sub

'========================================================================
' CUENTAS
'========================================================================

' Lo que paga quien entra (costo de DJ + percepción de Ingresos Brutos), los PB y el valor de venta.
' Falta: lo que no está en la lista de PSA (si no está vacío, la opción no se muestra).
Private Sub Calcular (o As OpcionIngreso, Kit As String, Percepcion As Double) As CuentaIngreso
	Dim c As CuentaIngreso
	c.Initialize
	c.Falta = ""
	Dim pct As Double = Percepcion
	Dim nombreKit As String = mapaKits.GetDefault(Kit, "")
	Dim costoKit As Double = ModCostos.Costo(nombreKit, "DJ", False)
	If costoKit <= 0 Then
		c.Falta = "Kit de Acceso c/" & Kit
	Else
		Dim iibbKit As Double = ModCostos.ListaSinIva(nombreKit) * pct / 100
		c.Contado = costoKit + iibbKit
		c.ContadoNoCat = ModCostos.Costo(nombreKit, "DJ", True) + iibbKit
	End If
	Dim ventaCompleta As Boolean = True
	For Each it As ItemIngreso In o.Items
		Dim costo As Double = ModCostos.Costo(it.Producto, "DJ", False)
		If costo <= 0 Then
			If c.Falta <> "" Then c.Falta = c.Falta & ", "
			c.Falta = c.Falta & it.Producto
			Continue
		End If
		Dim iibb As Double = ModCostos.ListaSinIva(it.Producto) * pct / 100
		c.Contado = c.Contado + it.Cantidad * (costo + iibb)
		c.ContadoNoCat = c.ContadoNoCat + it.Cantidad * (ModCostos.Costo(it.Producto, "DJ", True) + iibb)
		c.PuntosPB = c.PuntosPB + it.Cantidad * ModCostos.PB(it.Producto)
		Dim precio As Double = ModEquipos.PrecioActual(it.Producto)
		If precio <= 0 Then ventaCompleta = False
		c.Venta = c.Venta + it.Cantidad * precio
	Next
	If ventaCompleta = False Then c.Venta = 0       ' falta algún precio sugerido: no se muestra
	Return c
End Sub

' 9 PB o más: ingreso desarrollado (le falta poco para ser D) · 13 o más: entra como D
' Devuelve la clase de la etiqueta y el texto
Private Sub Categoria (PuntosPB As Double) As String()
	Dim centesimos As Long = Round(PuntosPB * 100)
	If centesimos >= 1300 Then Return Array As String("cD", "Entra como D")
	If centesimos >= 900 Then Return Array As String("cDES", "Ingreso desarrollado")
	Return Array As String("cDJ", "Entra como DJ")
End Sub

Private Sub TituloDe (o As OpcionIngreso, Kit As String) As String
	Return o.Titulo.Replace("{kit}", Kit)
End Sub

'========================================================================
' LA HOJA: A4 parada. Arriba las opciones con monotributo, abajo sin monotributo,
' y al final qué incluyen las destacadas (Promocionado, Pasa a Distribuidor).
'========================================================================

' Percepcion: % de Ingresos Brutos sobre el precio sin IVA (0 si no se cobra)
Public Sub ArmarHtml (Kit As String, Percepcion As Double) As String
	' Cada opción se calcula una vez (solo las que tienen todo en la lista de PSA)
	Dim validas As List
	validas.Initialize
	Dim cuentas As List
	cuentas.Initialize
	For Each o As OpcionIngreso In listaOpciones
		Dim c As CuentaIngreso = Calcular(o, Kit, Percepcion)
		If c.Falta = "" Then
			validas.Add(o)
			cuentas.Add(c)
		End If
	Next

	Dim sb As StringBuilder
	sb.Initialize
	sb.Append("<!doctype html><html><head><meta charset='utf-8'><meta name='viewport' content='width=794'>")
	sb.Append("<style>").Append(Estilos).Append("</style></head><body><div class='page'>")
	sb.Append("<div class='head'><div class='t'>OPCIONES DE INGRESO</div><div class='s'>con Kit de Acceso c/").Append(Esc(Kit)).Append("</div></div>")

	Bloque(sb, "MONOTRIBUTO O RESPONSABLE INSCRIPTO", False, validas, cuentas, Kit)
	Bloque(sb, "SIN MONOTRIBUTO", True, validas, cuentas, Kit)

	Dim cajas As StringBuilder
	cajas.Initialize
	For i = 0 To validas.Size - 1
		Dim op As OpcionIngreso = validas.Get(i)
		Dim cu As CuentaIngreso = cuentas.Get(i)
		If op.Destacada Then Caja(cajas, op, cu, Kit)
	Next
	If cajas.Length > 0 Then sb.Append("<div class='inc'>").Append(cajas.ToString).Append("</div>")

	sb.Append("<div class='nota'>")
	If notaArchivo <> "" Then
		sb.Append(Esc(notaArchivo))
	Else
		sb.Append("El Kit de Acceso trae las herramientas para trabajar (mochila, llaves, adaptadores, etc.) y se compra una sola vez. ")
		sb.Append("Ingreso desarrollado: 9&nbsp;PB o más. Para ser D: 13&nbsp;PB en un mes. 6 y 12 cuotas sin interés. ")
		sb.Append("Sin monotributo: con la percepción de IVA del 10,5&nbsp;%.")
	End If
	If Percepcion > 0 Then
		sb.Append(" Incluye percepción de Ingresos Brutos estimada: ").Append(ModGanancias.PBTexto(Percepcion))
		sb.Append("&nbsp;% sobre el precio sin IVA.")
	Else
		sb.Append(" No incluye percepción de Ingresos Brutos (depende de la provincia y de la inscripción).")
	End If
	sb.Append("</div>")

	sb.Append("<div class='foot'><div>")
	If ModCostos.Vigencia <> "" Then sb.Append("Precios de la lista de PSA vigente desde el ").Append(Esc(ModCostos.Vigencia)).Append(" · ")
	sb.Append("Valores orientativos.</div><div>Hecho con AquaPresu</div></div>")
	sb.Append("</div></body></html>")
	Return sb.ToString
End Sub

' Una tabla: Opción | Contado | 6 cuotas | 12 cuotas
Private Sub Bloque (sb As StringBuilder, Titulo As String, NoCat As Boolean, Validas As List, Cuentas As List, Kit As String)
	sb.Append("<div class='blq'><div class='tt'>").Append(Titulo).Append("</div>")
	sb.Append("<table><colgroup><col style='width:46%'><col><col><col></colgroup>")
	sb.Append("<thead><tr><th>Opción</th><th>Contado</th><th>6 cuotas</th><th>12 cuotas</th></tr></thead><tbody>")
	For i = 0 To Validas.Size - 1
		Dim o As OpcionIngreso = Validas.Get(i)
		Dim c As CuentaIngreso = Cuentas.Get(i)
		Dim total As Double = c.Contado
		If NoCat Then total = c.ContadoNoCat
		Dim cat() As String = Categoria(c.PuntosPB)
		If o.Destacada Then sb.Append("<tr class='pr'>") Else sb.Append("<tr>")
		sb.Append("<td class='o'>").Append(Esc(TituloDe(o, Kit)))
		sb.Append("<small>").Append(ModGanancias.PBTexto(c.PuntosPB)).Append("&nbsp;PB <span class='cat ").Append(cat(0)).Append("'>")
		sb.Append(cat(1)).Append("</span></small></td>")
		sb.Append("<td class='n tot'>").Append(Pesos(total)).Append("</td>")
		sb.Append("<td class='n c6'>").Append(Pesos(total / 6)).Append("</td>")
		sb.Append("<td class='n c12'>").Append(Pesos(total / 12)).Append("</td></tr>")
	Next
	sb.Append("</tbody></table></div>")
End Sub

' Recuadro de una opción destacada: qué lleva (uno por renglón, como en la planilla),
' cuánto le falta para ser D (o que entra como D) y el valor de venta de los equipos
Private Sub Caja (sb As StringBuilder, o As OpcionIngreso, c As CuentaIngreso, Kit As String)
	Dim cat() As String = Categoria(c.PuntosPB)
	sb.Append("<div class='b'><h4>").Append(Esc(TituloDe(o, Kit))).Append("</h4>")
	sb.Append("<ul><li><b>1</b>Kit de Acceso c/").Append(Esc(Kit)).Append("</li>")
	For Each it As ItemIngreso In o.Items
		sb.Append("<li><b>").Append(it.Cantidad).Append("</b>").Append(Esc(it.Nombre)).Append("</li>")
	Next
	sb.Append("</ul><div class='v'>")
	If cat(0) = "cDES" Then
		Dim faltan As Double = (1300 - Round(c.PuntosPB * 100)) / 100
		sb.Append("Le faltan <b>").Append(ModGanancias.PBTexto(faltan)).Append("&nbsp;PB</b> para ser D: 30 días para vender y reponer.<br>")
	Else If cat(0) = "cD" Then
		sb.Append("Con <b>").Append(ModGanancias.PBTexto(c.PuntosPB)).Append("&nbsp;PB</b> entra directo como D.<br>")
	End If
	If c.Venta > 0 Then sb.Append("Valor de venta de los equipos: <b>").Append(Pesos(c.Venta)).Append("</b>")
	sb.Append("</div></div>")
End Sub

'========================================================================
' FORMATOS
'========================================================================

' 930490.4 -> "$&nbsp;930.490" (el "$" no queda separado del número al cortar el renglón)
Private Sub Pesos (Valor As Double) As String
	Return ModAhorro.PesosTexto(Valor).Replace("$ ", "$&nbsp;")
End Sub

Private Sub Esc (Texto As String) As String
	Return Texto.Replace("&", "&amp;").Replace("<", "&lt;").Replace(">", "&gt;").Replace("'", "&#39;").Replace(QUOTE, "&quot;")
End Sub
